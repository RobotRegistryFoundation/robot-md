# robot-md v0.9.1 Implementation Plan — hybrid signing + RRF verification

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every robot registered with RRF carries a valid ML-DSA-65 + Ed25519 hybrid signature. robot-md always signs on register; RRF always verifies and rejects unsigned; auto-mint PATCH flow upgrades any unsigned-but-registered robot on its next command.

**Architecture:** Two-repo coordinated release. robot-md adds a `signing.py` module (wrapping `rcan.crypto.sign_hybrid` + Python `cryptography` for Ed25519), rewires `register.py` to sign every POST, and PATCHes RRF records that don't yet have signing keys. RRF adds `@noble/post-quantum` for ML-DSA-65 verify, a shared `functions/_lib/verify.ts` helper, a new PATCH/GET endpoint at `/v2/robots/[rrn]`, and tightens `register.ts` to reject unsigned. A Python-generated cross-language fixture is the CI gate both repos depend on.

**Tech Stack:** Python 3.10-3.13 · `rcan[pq,crypto]>=3.0,<4.0` · `cryptography>=42.0` · `dilithium-py>=1.0` · TypeScript · `@noble/post-quantum>=0.6.1` · vitest (RRF) · Cloudflare Pages Functions.

**Spec (authoritative):** `docs/superpowers/specs/2026-04-22-v0.9.1-hybrid-signing-design.md` at commit `1fb2e6b`. Do not widen.

**Budget:** ~3 sessions total.

```python
from cryptography.hazmat.primitives.asymmetric import ed25519
from rcan.crypto import generate_ml_dsa_keypair, sign_hybrid

# 1. Deterministic-ish keypairs (for reproducibility across runs we'd need a seeded RNG,
#    but interop is what matters, not determinism. Freshly generated keys each run.)
kp = generate_ml_dsa_keypair()
ed_sec = ed25519.Ed25519PrivateKey.generate()
ed_sec_bytes = ed_sec.private_bytes_raw()
ed_pub_bytes = ed_sec.public_key().public_bytes_raw()

# 2. Canonical body — must match what TS will canonicalize.
body = {"name": "spike-bob", "manufacturer": "acme", "model": "t1",
        "firmware_version": "1.0", "rcan_version": "3.0"}
message = json.dumps(body, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode("utf-8")

# 3. Hybrid sign.
hs = sign_hybrid(kp, ed_sec_bytes, message)

# 4. Compute pq_kid = first 8 hex of sha256(ml_dsa_pub).
ml_dsa_pub = kp.public_key_bytes
pq_kid = hashlib.sha256(ml_dsa_pub).hexdigest()[:8]

    "body": body,
    "canonical_bytes_b64": base64.b64encode(message).decode(),
    "pq_signing_pub": base64.b64encode(ml_dsa_pub).decode(),
    "pq_kid": pq_kid,

        "ml_dsa": base64.b64encode(hs.ml_dsa_signature).decode(),
        "ed25519": base64.b64encode(hs.ed25519_signature).decode(),

print(f"  ml_dsa_sig  {len(hs.ml_dsa_signature)} bytes")
```

```javascript
import { readFileSync } from "fs";
import { ml_dsa65 } from "@noble/post-quantum/ml-dsa";
import { webcrypto } from "crypto";

const fx = JSON.parse(readFileSync("/tmp/v091-spike/fixture.json", "utf8"));

function b64(s) { return Uint8Array.from(Buffer.from(s, "base64")); }

// 1. Recompute canonical bytes from the body dict — must equal fixture.canonical_bytes_b64.
function canonicalJson(obj) {
  const sort = (v) => {
    if (Array.isArray(v)) return v.map(sort);
    if (v && typeof v === "object") {
      const o = {};
      for (const k of Object.keys(v).sort()) o[k] = sort(v[k]);
      return o;
    }
    return v;
  };
  return new TextEncoder().encode(JSON.stringify(sort(obj)));
}

const msg = canonicalJson(fx.body);
const expected = b64(fx.canonical_bytes_b64);
if (Buffer.compare(Buffer.from(msg), Buffer.from(expected)) !== 0) {
  console.error("CANONICAL-JSON MISMATCH:");
  console.error("  python:", Buffer.from(expected).toString("utf8"));
  console.error("  ts:    ", Buffer.from(msg).toString("utf8"));
  process.exit(1);
}
console.log("CANONICAL-JSON OK");

// 2. Verify ML-DSA-65 signature.
const mlDsaOk = ml_dsa65.verify(b64(fx.pq_signing_pub), msg, b64(fx.sig.ml_dsa));
if (!mlDsaOk) { console.error("ML-DSA VERIFY FAILED"); process.exit(1); }
console.log("ML-DSA-65 VERIFY OK");

// 3. Verify Ed25519 signature via WebCrypto.
const edKey = await webcrypto.subtle.importKey(
  "raw", b64(fx.sig.ed25519_pub), { name: "Ed25519" }, false, ["verify"]
);
const edOk = await webcrypto.subtle.verify(
  "Ed25519", edKey, b64(fx.sig.ed25519), msg
);
if (!edOk) { console.error("ED25519 VERIFY FAILED"); process.exit(1); }
console.log("ED25519 VERIFY OK");
console.log("ALL GOOD — interop confirmed");
```

```typescript
import { ml_dsa65 } from "@noble/post-quantum/ml-dsa";

export type HybridSig = {
  ml_dsa: string;       // base64 ML-DSA-65 signature
  ed25519: string;      // base64 Ed25519 signature (64 bytes raw)
  ed25519_pub: string;  // base64 Ed25519 public key (32 bytes raw)
};

function sortKeys(v: unknown): unknown {
  if (Array.isArray(v)) return v.map(sortKeys);
  if (v && typeof v === "object") {
    const out: Record<string, unknown> = {};
    for (const k of Object.keys(v as Record<string, unknown>).sort()) {
      out[k] = sortKeys((v as Record<string, unknown>)[k]);
    }
    return out;
  }
  return v;
}

export function canonicalJson(obj: Record<string, unknown>): Uint8Array {
  return new TextEncoder().encode(JSON.stringify(sortKeys(obj)));
}

function b64(s: string): Uint8Array {
  return Uint8Array.from(Buffer.from(s, "base64"));
}

export async function verifyHybrid(
  ml_dsa_pub_b64: string,
  sig: HybridSig,
  message: Uint8Array,
): Promise<boolean> {
  try {
    // ML-DSA-65 via @noble/post-quantum
    const mlDsaOk = ml_dsa65.verify(b64(ml_dsa_pub_b64), message, b64(sig.ml_dsa));
    if (!mlDsaOk) return false;

    // Ed25519 via WebCrypto (both Node's webcrypto and Cloudflare Workers support it)
    const subtle = (globalThis.crypto ?? (await import("node:crypto")).webcrypto).subtle;
    const edKey = await subtle.importKey(
      "raw", b64(sig.ed25519_pub), { name: "Ed25519" }, false, ["verify"],
    );
    const edOk = await subtle.verify(
      "Ed25519", edKey, b64(sig.ed25519), message,
    );
    return edOk;
  } catch {
    return false;
  }
}
```

```typescript
import { canonicalJson, verifyHybrid } from "../../../_lib/verify.js";

export interface Env { RRF_KV: KVNamespace }

function ok(data: unknown, status = 200): Response {
  return new Response(JSON.stringify(data), {
    status, headers: { "Content-Type": "application/json" },
  });
}
function err(msg: string, status: number): Response {
  return new Response(JSON.stringify({ error: msg }), {
    status, headers: { "Content-Type": "application/json" },
  });
}

export const onRequestGet: PagesFunction<Env> = async ({ env, params }) => {
  const rrn = params.rrn as string;
  const raw = await env.RRF_KV.get(`robot:${rrn}`, "text");
  if (!raw) return err("Not found", 404);
  return ok(JSON.parse(raw));
};

export const onRequestPatch: PagesFunction<Env> = async ({ request, env, params }) => {
  const rrn = params.rrn as string;
  const auth = request.headers.get("Authorization");
  const apiKey = auth?.startsWith("Bearer ") ? auth.slice(7) : null;
  if (!apiKey) return err("Missing bearer token", 401);

  const raw = await env.RRF_KV.get(`robot:${rrn}`, "text");
  if (!raw) return err("Not found", 404);
  const record = JSON.parse(raw);
  if (record.api_key !== apiKey) return err("Unauthorized", 403);

  // v0.9.1: PATCH only upgrades null -> set.
  if (record.pq_signing_pub) {
    return err("Record already signed; key rotation not supported in v0.9.1", 409);
  }

  let body: {
    pq_signing_pub?: string; pq_kid?: string;
    sig?: { ml_dsa: string; ed25519: string; ed25519_pub: string };
  };
  try { body = await request.json() as typeof body; }
  catch { return err("Invalid JSON body", 400); }

  if (!body.pq_signing_pub || !body.pq_kid
      || !body.sig?.ml_dsa || !body.sig?.ed25519 || !body.sig?.ed25519_pub) {
    return err("Missing pq_signing_pub / pq_kid / sig", 400);
  }

  const message = canonicalJson({
    rrn, pq_signing_pub: body.pq_signing_pub, pq_kid: body.pq_kid,
  });
  const verified = await verifyHybrid(body.pq_signing_pub, body.sig, message);
  if (!verified) return err("Signature verification failed", 400);

  record.pq_signing_pub = body.pq_signing_pub;
  record.pq_kid = body.pq_kid;
  record.updated_at = new Date().toISOString();
  await env.RRF_KV.put(`robot:${rrn}`, JSON.stringify(record));
  return ok(record);
};
```

```python
"""ML-DSA-65 + Ed25519 hybrid signing for robot-md.

Thin wrapper around rcan.crypto (ML-DSA via dilithium-py) and python
cryptography (Ed25519) producing the signed-body shape that RRF's
functions/_lib/verify.ts expects.

Keystore layout: ~/.robot-md/keys/<rrn>.signing.json (mode 600), sibling
of the existing <rrn>.apikey.
"""

from __future__ import annotations

import base64
import hashlib
import json
import os
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from cryptography.hazmat.primitives.asymmetric import ed25519
from rcan.crypto import (
    HybridSignature,
    MlDsaKeyPair,
    generate_ml_dsa_keypair,
    sign_hybrid,
    verify_hybrid,
)

KEYSTORE_DIR = Path.home() / ".robot-md" / "keys"
KEY_FILE_MODE = 0o600
KEY_DIR_MODE = 0o700


@dataclass
class SigningKeypair:
    """ML-DSA-65 + Ed25519 keypair bundle for a single robot."""

    ml_dsa: MlDsaKeyPair
    ed25519_pub: bytes  # 32 bytes raw
    ed25519_sec: bytes  # 32 bytes raw
    pq_kid: str         # first 8 hex of sha256(ml_dsa.public_key_bytes)


def kid_from_pub(ml_dsa_pub: bytes) -> str:
    return hashlib.sha256(ml_dsa_pub).hexdigest()[:8]


def generate_keypair() -> SigningKeypair:
    ml_kp = generate_ml_dsa_keypair()
    ed_sec = ed25519.Ed25519PrivateKey.generate()
    return SigningKeypair(
        ml_dsa=ml_kp,
        ed25519_pub=ed_sec.public_key().public_bytes_raw(),
        ed25519_sec=ed_sec.private_bytes_raw(),
        pq_kid=kid_from_pub(ml_kp.public_key_bytes),
    )


def _keypath(rrn: str) -> Path:
    return KEYSTORE_DIR / f"{rrn}.signing.json"


def save_keypair(rrn: str, kp: SigningKeypair) -> Path:
    KEYSTORE_DIR.mkdir(parents=True, exist_ok=True)
    try:
        os.chmod(KEYSTORE_DIR, KEY_DIR_MODE)
    except OSError:
        pass  # best-effort; filesystem may not support (e.g., Windows)
    path = _keypath(rrn)
    data = {
        "rrn": rrn,
        "pq_kid": kp.pq_kid,
        "ml_dsa": {
            "pub": base64.b64encode(kp.ml_dsa.public_key_bytes).decode(),
            "sec": base64.b64encode(kp.ml_dsa.secret_key_bytes).decode(),
        },
        "ed25519": {
            "pub": base64.b64encode(kp.ed25519_pub).decode(),
            "sec": base64.b64encode(kp.ed25519_sec).decode(),
        },
        "created_at": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
    }
    # Write to tmp then rename to avoid partial writes; set mode 600 before content
    tmp = path.with_suffix(".tmp")
    tmp.write_text(json.dumps(data, indent=2))
    os.chmod(tmp, KEY_FILE_MODE)
    tmp.replace(path)
    return path


def load_keypair(rrn: str) -> SigningKeypair | None:
    path = _keypath(rrn)
    if not path.exists():
        return None
    data = json.loads(path.read_text())
    ml_pub = base64.b64decode(data["ml_dsa"]["pub"])
    ml_sec = base64.b64decode(data["ml_dsa"]["sec"])
    return SigningKeypair(
        ml_dsa=MlDsaKeyPair(public_key_bytes=ml_pub, secret_key_bytes=ml_sec),
        ed25519_pub=base64.b64decode(data["ed25519"]["pub"]),
        ed25519_sec=base64.b64decode(data["ed25519"]["sec"]),
        pq_kid=data["pq_kid"],
    )


def canonical_json(body: dict[str, Any]) -> bytes:
    """Deterministic JSON — must match TS `JSON.stringify(sortKeys(obj))`."""
    return json.dumps(
        body, sort_keys=True, separators=(",", ":"), ensure_ascii=False
    ).encode("utf-8")


def sign_body(kp: SigningKeypair, body: dict[str, Any]) -> dict[str, Any]:
    """Return a COPY of body with pq_signing_pub, pq_kid, sig appended.

    The signed message is canonical_json(body) *before* the three fields
    are attached. Verification re-strips sig and sort-encodes the rest,
    so ordering of appended fields is irrelevant.
    """
    message = canonical_json(body)
    hs = sign_hybrid(kp.ml_dsa, kp.ed25519_sec, message)
    return {
        **body,
        "pq_signing_pub": base64.b64encode(kp.ml_dsa.public_key_bytes).decode(),
        "pq_kid": kp.pq_kid,
        "sig": {
            "ml_dsa": base64.b64encode(hs.ml_dsa_signature).decode(),
            "ed25519": base64.b64encode(hs.ed25519_signature).decode(),
            "ed25519_pub": base64.b64encode(kp.ed25519_pub).decode(),
        },
    }


def verify_body(signed: dict[str, Any]) -> bool:
    """Strip sig/pq_signing_pub/pq_kid, canonicalize, hybrid-verify.

    Note: for the sign-register path the verifier needs to know
    pq_signing_pub rode in the signed fields (it did) — after stripping
    sig we still have pq_signing_pub in `rest`, so the canonical bytes
    include it. Matches RRF's verify step which includes pq_signing_pub
    in the signed_fields dict.
    """
    try:
        sig = signed.get("sig")
        if not sig:
            return False
        rest = {k: v for k, v in signed.items() if k != "sig"}
        pq_pub_b64 = rest.get("pq_signing_pub")
        if not pq_pub_b64:
            return False
        message = canonical_json(rest)
        verify_hybrid(
            ml_dsa_public_key_bytes=base64.b64decode(pq_pub_b64),
            ed25519_public_key_bytes=base64.b64decode(sig["ed25519_pub"]),
            message=message,
            hybrid_sig=HybridSignature(
                ml_dsa_signature=base64.b64decode(sig["ml_dsa"]),
                ed25519_signature=base64.b64decode(sig["ed25519"]),
            ),
        )
        return True
    except Exception:
        return False
```

**Subtlety — sign_body vs. verify_body.** `sign_body` canonicalizes BEFORE attaching sig/pq_signing_pub/pq_kid. `verify_body` canonicalizes AFTER stripping only sig — so pq_signing_pub and pq_kid ARE included in the verified message. But `sign_body` didn't include them in the signed message.

Fix: `sign_body` MUST include pq_signing_pub and pq_kid in the canonical-JSON message before signing. Rewrite:

```python
def sign_body(kp: SigningKeypair, body: dict[str, Any]) -> dict[str, Any]:
    pq_pub_b64 = base64.b64encode(kp.ml_dsa.public_key_bytes).decode()
    body_with_ids = {**body, "pq_signing_pub": pq_pub_b64, "pq_kid": kp.pq_kid}
    message = canonical_json(body_with_ids)
    hs = sign_hybrid(kp.ml_dsa, kp.ed25519_sec, message)
    return {
        **body_with_ids,
        "sig": {
            "ml_dsa": base64.b64encode(hs.ml_dsa_signature).decode(),
            "ed25519": base64.b64encode(hs.ed25519_signature).decode(),
            "ed25519_pub": base64.b64encode(kp.ed25519_pub).decode(),
        },
    }
```

This matches verify_body's stripping + re-canonicalize path AND matches RRF's register.ts which signs over `{...fields, pq_signing_pub, pq_kid}`.

```python
"""RURI construction for robot-md register.

Spec-mandated format: rcan://<registry-host>/<manufacturer>/<model>/<device-id>.
robot-md v0.9.1 defaults <registry-host> to robotregistryfoundation.org.
"""

from __future__ import annotations

import re
from typing import Any

REGISTRY_HOST = "robotregistryfoundation.org"

_SLUG_RE = re.compile(r"[^a-z0-9]+")


def slug(s: str) -> str:
    """Lowercase, alphanumeric-or-hyphen, collapse repeats, strip ends.

    Raises ValueError if the result is empty (all non-alphanumeric input).
    """
    out = _SLUG_RE.sub("-", s.lower()).strip("-")
    if not out:
        raise ValueError(f"cannot slugify {s!r} — no alphanumeric characters")
    return out


def construct_ruri(manifest: dict[str, Any]) -> str:
    """Return the RURI for this manifest.

    If metadata.ruri is explicitly set, return it verbatim.
    Otherwise build rcan://{REGISTRY_HOST}/{slug(manufacturer)}/{slug(model)}/{slug(robot_name)}.
    """
    meta = manifest.get("metadata") or {}
    explicit = (meta.get("ruri") or "").strip()
    if explicit:
        return explicit

    for k in ("robot_name", "manufacturer", "model"):
        if not meta.get(k):
            raise ValueError(f"cannot construct RURI: metadata.{k} missing")

    return (
        f"rcan://{REGISTRY_HOST}"
        f"/{slug(meta['manufacturer'])}"
        f"/{slug(meta['model'])}"
        f"/{slug(meta['robot_name'])}"
    )
```

```python
def patch_rrf(
    url: str, body: dict[str, Any], api_key: str, *, timeout: float = 15.0,
) -> dict[str, Any]:
    """PATCH the RRF record with the signing-key upgrade."""
    req = urllib.request.Request(
        url, method="PATCH",
        headers={
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        },
        data=json.dumps(body).encode(),
    )
    with urllib.request.urlopen(req, timeout=timeout) as r:
        return json.loads(r.read())


def auto_mint_if_needed(rrn: str, *, endpoint: str = DEFAULT_ENDPOINT) -> None:
    """If this RRN has an apikey but no signing keypair, generate + PATCH.

    Silently no-ops if either (a) a signing keypair already exists, or
    (b) no apikey exists (caller will surface its own auth error later).
    """
    from robot_md.signing import (
        canonical_json, generate_keypair, load_keypair, save_keypair, sign_body,
    )

    if load_keypair(rrn) is not None:
        return
    api_key = load_apikey(rrn)
    if not api_key:
        return

    kp = generate_keypair()
    patch_body_unsigned = {
        "rrn": rrn,
        "pq_signing_pub": base64.b64encode(kp.ml_dsa.public_key_bytes).decode(),
        "pq_kid": kp.pq_kid,
    }
    signed = sign_body(kp, patch_body_unsigned)
    # sign_body added pq_signing_pub + pq_kid + sig; the PATCH endpoint expects
    # only pq_signing_pub + pq_kid + sig (no rrn in the HTTP body — rrn is the URL).
    patch_http_body = {
        "pq_signing_pub": signed["pq_signing_pub"],
        "pq_kid": signed["pq_kid"],
        "sig": signed["sig"],
    }
    # Base URL of register endpoint is .../v2/robots/register; PATCH is .../v2/robots/<rrn>.
    # Derive:
    patch_url = endpoint.replace("/register", f"/{rrn}")
    try:
        patch_rrf(patch_url, patch_http_body, api_key)
    except urllib.error.HTTPError as e:
        raise RuntimeError(
            f"auto-mint PATCH failed for {rrn}: {e.code} {e.read().decode(errors='replace')}"
        ) from e

    save_keypair(rrn, kp)
```

**Wait — the PATCH payload subtlety.** RRF verifies over canonical `{rrn, pq_signing_pub, pq_kid}` bytes. `sign_body` canonicalizes over the body dict PLUS pq_signing_pub and pq_kid (which it adds). If we pass `{rrn, pq_signing_pub, pq_kid}` as input, `sign_body` would effectively sign `{rrn, pq_signing_pub, pq_kid, pq_signing_pub, pq_kid}` (duplicate keys). That doesn't work.

Fix: for the PATCH case, sign directly without going through `sign_body`:

```python
def auto_mint_if_needed(rrn: str, *, endpoint: str = DEFAULT_ENDPOINT) -> None:
    from rcan.crypto import sign_hybrid
    from robot_md.signing import (
        canonical_json, generate_keypair, load_keypair, save_keypair,
    )
    import base64

    if load_keypair(rrn) is not None:
        return
    api_key = load_apikey(rrn)
    if not api_key:
        return

    kp = generate_keypair()
    ml_pub_b64 = base64.b64encode(kp.ml_dsa.public_key_bytes).decode()

    # RRF verifies over canonical {rrn, pq_signing_pub, pq_kid}.
    signed_payload = {"rrn": rrn, "pq_signing_pub": ml_pub_b64, "pq_kid": kp.pq_kid}
    message = canonical_json(signed_payload)
    hs = sign_hybrid(kp.ml_dsa, kp.ed25519_sec, message)

    patch_body = {
        "pq_signing_pub": ml_pub_b64,
        "pq_kid": kp.pq_kid,
        "sig": {
            "ml_dsa": base64.b64encode(hs.ml_dsa_signature).decode(),
            "ed25519": base64.b64encode(hs.ed25519_signature).decode(),
            "ed25519_pub": base64.b64encode(kp.ed25519_pub).decode(),
        },
    }

    patch_url = endpoint.replace("/register", f"/{rrn}")
    try:
        patch_rrf(patch_url, patch_body, api_key)
    except urllib.error.HTTPError as e:
        raise RuntimeError(
            f"auto-mint PATCH failed for {rrn}: {e.code} {e.read().decode(errors='replace')}"
        ) from e

    save_keypair(rrn, kp)
```
